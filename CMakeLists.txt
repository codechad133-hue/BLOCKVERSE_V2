cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")  # static CRT: no VC++ redistributable needed
project(BLOCKVERSE VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
# ENet's own CMakeLists declares an old minimum version; CMake 4+ refuses that without this.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5)
FetchContent_MakeAvailable(raylib)
FetchContent_Declare(enet GIT_REPOSITORY https://github.com/lsalzman/enet.git GIT_TAG v1.3.18)
FetchContent_MakeAvailable(enet)
add_executable(BLOCKVERSE src/main.cpp)
target_link_libraries(BLOCKVERSE PRIVATE raylib enet)
# ENet only sets its include path directory-wide, not on the target, so add it explicitly.
target_include_directories(BLOCKVERSE PRIVATE ${enet_SOURCE_DIR}/include)
if(WIN32)
  target_link_libraries(BLOCKVERSE PRIVATE winmm)
endif()
if(MSVC)
  target_compile_options(BLOCKVERSE PRIVATE /O2 /W4 /permissive-)
  target_compile_definitions(BLOCKVERSE PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(BLOCKVERSE PRIVATE -O2 -Wall -Wextra)
endif()

install(TARGETS BLOCKVERSE RUNTIME DESTINATION .)
install(FILES LICENSE.txt README.md DESTINATION .)
